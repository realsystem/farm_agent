DEBUGGING
────────────────────────────────────────────────────────────────────────────────

View logs (follow mode)
  docker-compose -f docker-compose.dev.yml logs -f farm-agent

View last 20 lines
  docker-compose -f docker-compose.dev.yml logs farm-agent --tail 20

Check containers
  docker-compose -f docker-compose.dev.yml ps

Restart agent
  docker-compose -f docker-compose.dev.yml restart farm-agent

Check Home Assistant entities
  HA_TOKEN=$(grep HA_TOKEN= .env | cut -d= -f2)
  curl -s 'http://localhost:8123/api/states' \
    -H "Authorization: Bearer $HA_TOKEN" | python3 -m json.tool

Test OpenAI connectivity
  docker exec farm-agent python3 -c \
    "import socket; s=socket.socket(); print('✓ OK' if s.connect_ex(('api.openai.com', 443))==0 else '✗ BLOCKED')"

View Home Assistant logs
  docker-compose -f docker-compose.dev.yml logs homeassistant --tail 20
